## Question 19: How do you validate form inputs and insert valid values into MySQL?
**Answer:**
```python
import re, pymysql
from dbconfig import DB

def clean_user(name, email, age):
    name = name.strip()
    email = email.strip().lower()
    if not re.fullmatch(r"[A-Za-z ]{2,50}", name):
        raise ValueError("Invalid name")
    if not re.fullmatch(r"[^@\s]+@[^@\s]+\.[a-z]{2,}", email, re.I):
        raise ValueError("Invalid email")
    age = int(age)
    if not 1 <= age <= 120:
        raise ValueError("Invalid age")
    return name, email, age

try:
    vals = clean_user("  Rahul ", "rahul@example.com", "24")
    with pymysql.connect(**DB) as conn:
        with conn.cursor() as cur:
            cur.execute("INSERT INTO users (name, email, age) VALUES (%s, %s, %s)", vals)
        conn.commit()
    print("Saved", vals)
except ValueError as e:
    print("Rejected:", e)
```
