## Question 2: How do you determine whether an integer is prime, composite, or neither?
**Answer:**
```html
<input type="number" id="n"> <button onclick="check()">Check</button>
<div id="out"></div>
<script>
  function check() {
    const n = Number(document.getElementById("n").value);
    if (!Number.isInteger(n)) {
      document.getElementById("out").textContent = "Enter a valid integer.";
      return;
    }
    let prime = n > 1;
    for (let i = 2; i * i <= n; i++) if (n % i === 0) { prime = false; break; }
    document.getElementById("out").innerText = n < 2 ? n + " is neither prime nor composite" : n + (prime ? " is Prime" : " is Composite");
  }
</script>
```
