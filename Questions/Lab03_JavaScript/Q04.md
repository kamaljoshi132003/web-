## Question 4: How can you generate the Fibonacci sequence using `for`, `while`, and `do...while` loops?
**Answer:**
```html
<script>
  let n = 10, out = "";
  // for
  let a = 0, b = 1; out += "for: ";
  for (let i = 0; i < n; i++) { out += a + " "; [a, b] = [b, a + b]; }
  // while
  a = 0; b = 1; let i = 0; out += "<br>while: ";
  while (i < n) { out += a + " "; [a, b] = [b, a + b]; i++; }
  // do-while
  a = 0; b = 1; i = 0; out += "<br>do-while: ";
  do { out += a + " "; [a, b] = [b, a + b]; i++; } while (i < n);
  document.write(out);
</script>
```
