## Question 3: How do you assign a grade with conditional statements and select a remark with a `switch` statement?
**Answer:**
```html
<script>
  const marks = Number.parseInt(prompt("Enter marks (0-100):") ?? "", 10);
  if (!Number.isInteger(marks) || marks < 0 || marks > 100) {
    document.write("Enter a whole number from 0 to 100.");
  } else {
  // if-else
  let g;
  if (marks >= 90) g = "A+"; else if (marks >= 80) g = "A"; else if (marks >= 70) g = "B";
  else if (marks >= 60) g = "C"; else if (marks >= 40) g = "D"; else g = "F";
  // switch
  let remark;
  switch (g) { case "A+": case "A": remark = "Excellent"; break; case "B": case "C": remark = "Good"; break;
               case "D": remark = "Pass"; break; default: remark = "Fail"; }
  document.write(`Grade: ${g} (${remark})`);
  }
</script>
```
