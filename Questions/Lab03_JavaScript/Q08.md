## Question 8: How do you build a basic calculator whose buttons use `addEventListener`?
**Answer:**
```html
<input id="d" value="0" readonly aria-label="Calculator display"><br>
<div id="keys"></div>
<script>
  const keys = ["7","8","9","/","4","5","6","*","1","2","3","-","0",".","=","+","C"];
  const box = document.getElementById("keys");
  const display = document.getElementById("d");
  let current = "0";
  let previous = null;
  let operator = null;
  let resetCurrent = false;

  function calculate(left, right, operation) {
    if (operation === "+") return left + right;
    if (operation === "-") return left - right;
    if (operation === "*") return left * right;
    if (operation === "/") return right === 0 ? null : left / right;
    return right;
  }

  keys.forEach(k => {
    const b = document.createElement("button");
    b.type = "button";
    b.textContent = k;
    box.appendChild(b);
    b.addEventListener("click", () => {
      if (k === "C") {
        current = "0";
        previous = null;
        operator = null;
        resetCurrent = false;
      } else if (["+", "-", "*", "/"].includes(k)) {
        if (previous !== null && operator && !resetCurrent) {
          const result = calculate(previous, Number(current), operator);
          if (result === null || !Number.isFinite(result)) {
            current = "Error";
            previous = null;
            operator = null;
            display.value = current;
            return;
          }
          current = String(result);
        }
        previous = Number(current);
        operator = k;
        resetCurrent = true;
      } else if (k === "=") {
        if (previous !== null && operator) {
          const result = calculate(previous, Number(current), operator);
          current = result === null || !Number.isFinite(result) ? "Error" : String(result);
          previous = null;
          operator = null;
          resetCurrent = true;
        }
      } else if (k === ".") {
        if (resetCurrent || current === "Error") current = "0";
        if (!current.includes(".")) current += ".";
        resetCurrent = false;
      } else {
        if (resetCurrent || current === "Error") current = k;
        else current = current === "0" ? k : current + k;
        resetCurrent = false;
      }
      display.value = current;
    });
  });
</script>
```
