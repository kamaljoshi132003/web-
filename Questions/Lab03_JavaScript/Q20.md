## Question 20: How do you generate a table using row and column counts entered by the user?
**Answer:**
```html
Rows: <input type="number" id="r" value="3"> Cols: <input type="number" id="c" value="3">
<button onclick="gen()">Generate</button>
<div id="tbl"></div>
<script>
  function gen() {
    const rows = Number(document.getElementById("r").value);
    const columns = Number(document.getElementById("c").value);
    const container = document.getElementById("tbl");
    container.replaceChildren();
    if (!Number.isInteger(rows) || !Number.isInteger(columns) || rows < 1 || columns < 1 || rows > 50 || columns > 50) {
      container.textContent = "Enter row and column counts from 1 to 50.";
      return;
    }
    const table = document.createElement("table");
    table.border = "1";
    table.cellPadding = "6";
    for (let row = 1; row <= rows; row++) {
      const tableRow = table.insertRow();
      for (let column = 1; column <= columns; column++) {
        tableRow.insertCell().textContent = `R${row}C${column}`;
      }
    }
    container.appendChild(table);
  }
</script>
```
