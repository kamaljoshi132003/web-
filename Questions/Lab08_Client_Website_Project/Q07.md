## Question 7: How do you create a booking form and validate its fields with JavaScript?
**Answer:** Save the following example in `booking.html`.
```html
<form id="bookForm" novalidate>
  <label for="bn">Name</label><input id="bn" name="name" required>
  <label for="be">Email</label><input id="be" name="email" type="email" required>
  <label for="bp">Phone</label><input id="bp" name="phone" type="tel" required>
  <label for="bd">Date</label><input id="bd" name="date" type="date" required>
  <label for="bg">Guests</label>
  <select id="bg" name="guests"><option>1</option><option>2</option><option>4</option><option>6</option></select>
  <button class="btn" type="submit">Reserve</button>
  <p id="bmsg" role="alert"></p>
</form>
<script>
document.getElementById("bookForm").addEventListener("submit", e => {
  e.preventDefault();
  const msg = document.getElementById("bmsg");
  const nameInput = document.getElementById("bn");
  const emailInput = document.getElementById("be");
  const phoneInput = document.getElementById("bp");
  const dateInput = document.getElementById("bd");
  const guestInput = document.getElementById("bg");
  const name = nameInput.value.trim();
  const mail = emailInput.value.trim();
  const phone = phoneInput.value.trim();
  const date = dateInput.value;
  if (name.length < 2) return msg.textContent = "Please enter your name.";
  if (!/^[^@\s]+@[^@\s]+\.[a-z]{2,}$/i.test(mail)) return msg.textContent = "Enter a valid email.";
  if (!/^[0-9]{10}$/.test(phone)) return msg.textContent = "Phone must contain 10 digits.";
  const today = new Date();
  const localToday = `${today.getFullYear()}-${String(today.getMonth() + 1).padStart(2, "0")}-${String(today.getDate()).padStart(2, "0")}`;
  if (!date || date < localToday) return msg.textContent = "Choose today or a future date.";
  msg.style.color = "green";
  msg.textContent = `Thank you, ${name}. Your booking request for ${guestInput.value} guest(s) on ${date} has been received.`;
});
</script>
```
