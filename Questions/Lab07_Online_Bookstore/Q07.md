## Question 7: How do you validate checkout details and record an order and its items in the database?
**Answer:** Save the following code as `checkout.php`.
```php
<?php
require_once __DIR__ . '/db.php';
require_login();
$cart = $_SESSION['cart'] ?? [];
if (!$cart) die("Cart is empty");
$checkoutError = "";
if ($_SERVER['REQUEST_METHOD'] === 'POST') {
  $name = trim($_POST['name'] ?? '');
  $address = trim($_POST['address'] ?? '');
  $city = trim($_POST['city'] ?? '');
  $pin = trim($_POST['pin'] ?? '');
  if ($name === '' || $address === '' || $city === '' || !preg_match('/^[0-9]{6}$/', $pin)) {
    $checkoutError = "Enter all shipping details and a valid six-digit PIN code.";
  } else {
    $addr = "$name, $address, $city - $pin";
    try {
      $pdo->beginTransaction();
      $total = 0; $items = [];
      foreach ($cart as $id => $qty) {
        $s = $pdo->prepare("SELECT price, stock FROM books WHERE id = ? FOR UPDATE"); $s->execute([$id]); $b = $s->fetch();
        if (!$b || $b['stock'] < $qty) throw new Exception("Not enough stock for book #$id");
        $total += $b['price'] * $qty; $items[] = [$id, $qty, $b['price']];
      }
      $pdo->prepare("INSERT INTO orders (user_id, total, address) VALUES (?, ?, ?)")->execute([$_SESSION['uid'], $total, $addr]);
      $oid = $pdo->lastInsertId();
      foreach ($items as [$id, $qty, $price]) {
        $pdo->prepare("INSERT INTO order_items (order_id, book_id, qty, price) VALUES (?, ?, ?, ?)")->execute([$oid, $id, $qty, $price]);
        $pdo->prepare("UPDATE books SET stock = stock - ? WHERE id = ?")->execute([$qty, $id]);
      }
      $pdo->commit();
      unset($_SESSION['cart']);
      header("Location: invoice.php?id=$oid"); exit;
    } catch (Exception $ex) {
      if ($pdo->inTransaction()) $pdo->rollBack();
      $checkoutError = $ex->getMessage();
        }
  }
}
?>
<?php include 'header.php'; ?>
<form method="post" id="checkoutForm" style="padding:24px"><h2>Shipping Details</h2>
  <?php if ($checkoutError !== ''): ?><p role="alert" style="color:red"><?= e($checkoutError) ?></p><?php endif; ?>
  <input name="name" placeholder="Full name" required><br><input name="address" placeholder="Address" required><br>
  <input name="city" placeholder="City" required><br><input name="pin" placeholder="PIN code" inputmode="numeric" pattern="[0-9]{6}" required><br>
  <button>Place Order</button></form>
<script src="validate.js"></script>
<?php include 'footer.php'; ?>
```
