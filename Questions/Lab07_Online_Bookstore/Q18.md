## Question 18: How do you add client-side validation to registration, login, and checkout forms?
**Answer:** Save the following code as `validate.js`.
```javascript
function bad(msg) { alert(msg); return false; }
document.addEventListener("DOMContentLoaded", () => {
  const reg = document.getElementById("regForm");
  if (reg) reg.onsubmit = () => {
    if (reg.elements.namedItem("name").value.trim().length < 2) return bad("Enter a valid name");
    if (!/^[^@\s]+@[^@\s]+\.[a-z]{2,}$/i.test(reg.elements.namedItem("email").value)) return bad("Enter a valid email");
    if (reg.elements.namedItem("password").value.length < 6) return bad("Password must be at least 6 characters");
  };
  const login = document.getElementById("loginForm");
  if (login) login.onsubmit = () => {
    if (!login.elements.namedItem("email").value || !login.elements.namedItem("password").value) return bad("Email and password are required.");
  };
  const co = document.getElementById("checkoutForm");
  if (co) co.onsubmit = () => {
    if (!co.elements.namedItem("name").value.trim() || !co.elements.namedItem("address").value.trim() || !co.elements.namedItem("city").value.trim()) return bad("Enter all shipping details.");
    if (!/^[0-9]{6}$/.test(co.elements.namedItem("pin").value)) return bad("The PIN code must contain six digits.");
  };
});
```
Client-side validation improves usability but can be bypassed. Always validate submitted data on the server as well.
